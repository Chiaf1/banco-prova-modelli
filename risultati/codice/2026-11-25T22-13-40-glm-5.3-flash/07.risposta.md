export function aRomano(n: number): string {
  if (!Number.isInteger(n) || n < 1 || n > 3999) {
    throw new RangeError("numero non valido");
  }
  const migliaia = ["", "M", "MM", "MMM"];
  const centinaia = ["", "C", "CC", "CCC", "CD", "D", "DC", "DCC", "DCCC", "CM"];
  const decine = ["", "X", "XX", "XXX", "XL", "L", "LX", "LXX", "LXXX", "XC"];
  const unita = ["", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX"];
  const m = Math.floor(n / 1000);
  const c = Math.floor((n % 1000) / 100);
  const d = Math.floor((n % 100) / 10);
  const u = n % 10;
  return migliaia[m] + centinaia[c] + decine[d] + unita[u];
}

const ROMANO_STANDARD = /^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$/;

export function daRomano(s: string): number {
  if (!ROMANO_STANDARD.test(s) || s === "") {
    throw new RangeError("numero romano non valido");
  }
  const valori: Record<string, number> = {
    I: 1, V: 5, X: 10, L: 50, C: 100, D: 500, M: 1000,
  };
  let tot = 0;
  for (let i = 0; i < s.length; i++) {
    const v = valori[s[i]]!;
    if (i + 1 < s.length && valori[s[i + 1]!]! > v) {
      tot -= v;
    } else {
      tot += v;
    }
  }
  return tot;
}
