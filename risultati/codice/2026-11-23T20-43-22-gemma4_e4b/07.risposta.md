/**
 * Converte un intero (1-3999) nel suo equivalente in numeri romani standard.
 * @param n L'intero da convertire.
 * @returns La stringa del numero romano.
 * @throws {RangeError} Se n non è un intero tra 1 e 3999.
 */
export function aRomano(n: number): string {
  if (!Number.isInteger(n) || n < 1 || n > 3999) {
    throw new RangeError("Input must be an integer between 1 and 3999.");
  }

  const values = [
    [1000, 'M'],
    [900, 'CM'],
    [500, 'D'],
    [400, 'CD'],
    [100, 'C'],
    [90, 'XC'],
    [50, 'L'],
    [40, 'XL'],
    [10, 'X'],
    [9, 'IX'],
    [5, 'V'],
    [4, 'IV'],
    [1, 'I'],
  ];

  let result = '';
  let num = n;

  for (const [value, symbol] of values) {
    while (num >= value) {
      result += symbol;
      num -= value;
    }
  }

  return result;
}

/**
 * Converte una stringa di numeri romani standard in un intero.
 * @param s La stringa del numero romano.
 * @returns L'intero corrispondente.
 * @throws {RangeError} Se la stringa non è un numero romano standard valido.
 */
export function daRomano(s: string): number {
  if (s.length === 0) {
    throw new RangeError("Input string cannot be empty.");
  }

  // Regex che garantisce che la stringa sia una rappresentazione standard (1-3999).
  // M{0,3} -> 0-3 migliaia
  // (CM|CD|D?C{0,3}) -> Centinaia (CM, CD, o D seguito da 0-3 C)
  // (XC|XL|L?X{0,3}) -> Decine (XC, XL, o L seguito da 0-3 X)
  // (IX|IV|V?I{0,3}) -> Unità (IX, IV, o V seguito da 0-3 I)
  const romanRegex = /^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$/;

  if (!romanRegex.test(s)) {
    throw new RangeError("Input string is not a valid standard Roman numeral.");
  }

  const values = {
    'I': 1,
    'V': 5,
    'X': 10,
    'L': 50,
    'C': 100,
    'D': 500,
    'M': 1000,
  };

  let result = 0;

  for (let i = 0; i < s.length; i++) {
    const currentVal = values[s[i]];
    const nextVal = values[s[i + 1]];

    // Se il valore corrente è minore del valore successivo (caso sottrattivo: IV, IX, etc.)
    if (nextVal && currentVal < nextVal) {
      result += nextVal - currentVal;
      i++; // Salta il carattere successivo
    } else {
      result += currentVal;
    }
  }

  return result;
}
