/**
 * Riceve intervalli chiusi [inizio, fine] e restituisce gli intervalli fusi.
 *
 * @param intervalli L'array di intervalli da fondere.
 * @returns Un array di intervalli fusi, ordinati e non sovrapposti.
 * @throws {RangeError} Se un intervallo è invalido (valori non finiti o inizio > fine).
 */
export function fondiIntervalli(intervalli: [number, number][]): [number, number][] {
  // 1. Validazione dell'input
  for (const [start, end] of intervalli) {
    if (!isFinite(start) || !isFinite(end) || start > end) {
      throw new RangeError("Invalid interval provided: values must be finite and start <= end.");
    }
  }

  // Input vuoto
  if (intervalli.length === 0) {
    return [];
  }

  // 2. Ordinamento: Ordina gli intervalli in base al loro punto di inizio.
  // Creiamo una copia per non modificare l'input.
  const sortedIntervals = [...intervalli].sort((a, b) => a[0] - b[0]);

  // 3. Logica di fusione
  const merged: [number, number][] = [];

  // Inizializziamo con il primo intervallo
  let currentStart = sortedIntervals[0][0];
  let currentEnd = sortedIntervals[0][1];

  for (let i = 1; i < sortedIntervals.length; i++) {
    const nextInterval = sortedIntervals[i];
    const nextStart = nextInterval[0];
    const nextEnd = nextInterval[1];

    // Se l'intervallo successivo si sovrappone o tocca l'intervallo corrente
    // (nextStart <= currentEnd)
    if (nextStart <= currentEnd) {
      // Fusi: aggiorniamo la fine dell'intervallo corrente al massimo tra le due fine
      currentEnd = Math.max(currentEnd, nextEnd);
    } else {
      // Non si sovrappongono: aggiungiamo l'intervallo fuso precedente al risultato
      merged.push([currentStart, currentEnd]);

      // Iniziamo un nuovo intervallo fuso
      currentStart = nextStart;
      currentEnd = nextEnd;
    }
  }

  // Aggiungiamo l'ultimo intervallo fuso al risultato
  merged.push([currentStart, currentEnd]);

  return merged;
}
