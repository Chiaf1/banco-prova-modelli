/**
 * Valuta espressioni aritmetiche.
 * 
 * Utilizza l'algoritmo di valutazione basato su due stack (operandi e operatori)
 * per gestire la precedenza e le parentesi.
 */

type Operator = '+' | '-' | '*' | '/' | 'U+' | 'U-';

/**
 * Definisce la precedenza degli operatori.
 * Più alto è il valore, maggiore è la precedenza.
 */
function precedence(op: Operator): number {
  switch (op) {
    case '+' :
    case '-' :
      return 1;
    case '*' :
    case '/' :
      return 2;
    default:
      return 0;
  }
}

/**
 * Esegue l'operazione data dall'operatore.
 * @param op L'operatore.
 * @param b Il secondo operando.
 * @param a Il primo operando.
 * @returns Il risultato dell'operazione.
 */
function applyOp(op: Operator, b: number, a: number): number {
  switch (op) {
    case '+':
      return a + b;
    case '-':
      return a - b;
    case '*':
      return a * b;
    case '/':
      // Divisione segue JavaScript: 1 / 0 vale Infinity
      return a / b;
    case 'U+':
      return b; // Unary plus does nothing
    case 'U-':
      return -b; // Unary minus negates the operand
    default:
      throw new SyntaxError("Operatore sconosciuto");
  }
}

/**
 * Valuta l'espressione aritmetica fornita come stringa.
 * @param espressione La stringa dell'espressione.
 * @returns Il valore numerico dell'espressione.
 * @throws SyntaxError se l'espressione è malformata.
 */
export function valuta(espressione: string): number {
  const tokens: string[] = [];
  let i = 0;
  const len = espressione.length;

  // --- Fase 1: Tokenizzazione e Pre-elaborazione ---
  // Raccoglie numeri, operatori e parentesi.
  // Gestisce i numeri decimali e i segni unari/binari.
  while (i < len) {
    let char = espressione[i];

    // Ignora gli spazi
    if (char === ' ') {
      i++;
      continue;
    }

    // Numeri (interi o decimali)
    if (/[0-9.]/.test(char)) {
      let start = i;
      let hasDecimal = false;
      
      // Trova la fine del numero
      while (i < len && /[0-9.]/.test(espressione[i])) {
        if (espressione[i] === '.') {
          if (hasDecimal) {
            throw new SyntaxError("Numero malformato: più di un punto decimale.");
          }
          hasDecimal = true;
        }
        i++;
      }
      
      const numStr = espressione.substring(start, i);
      
      // Validazione finale del numero (es. non può essere solo ".")
      if (numStr === '.' || numStr === '') {
          throw new SyntaxError("Numero malformato.");
      }

      tokens.push(parseFloat(numStr));
      continue;
    }

    // Operatori e Parentesi
    if (['+', '-', '*', '/', '(', ')'].includes(char)) {
      // Gestione dei segni unari/binari
      let token: string | Operator = char;
      
      // Un operatore è unario se:
      // 1. È il primo carattere dell'espressione.
      // 2. Segue una parentesi aperta '('.
      // 3. Segue un altro operatore binario.
      const isUnaryContext = (
        i === 0 || 
        espressione[i - 1] === '(' || 
        ['+', '-', '*', '/'].includes(espressione[i - 1])
      );

      if ((char === '+' || char === '-') && isUnaryContext) {
        // Se è un contesto unario, lo trattiamo come un operatore speciale
        token = char === '+' ? 'U+' : 'U-';
      }
      
      tokens.push(token as any);
      i++;
      continue;
    }

    // Carattere sconosciuto
    throw new SyntaxError(`Carattere sconosciuto: ${char}`);
  }

  // --- Fase 2: Valutazione (Algoritmo Shunting-Yard semplificato) ---
  
  const values: number[] = [];
  const ops: Operator[] = [];
  let tokenIndex = 0;

  while (tokenIndex < tokens.length) {
    const token = tokens[tokenIndex];

    if (typeof token === 'number') {
      // È un operando
      values.push(token);
    } else if (typeof token === 'string' && ['(', ')'].includes(token)) {
      // È una parentesi
      if (token === '(') {
        ops.push(token as Operator);
      } else { // token === ')'
        // Esegui tutte le operazioni finché non incontra '('
        while (ops.length > 0 && ops[ops.length - 1] !== '(') {
          const op = ops.pop() as Operator;
          if (values.length < 2) throw new SyntaxError("Operandi mancanti.");
          const val2 = values.pop()!;
          const val1 = values.pop()!;
          values.push(applyOp(op, val2, val1));
        }
        
        // Rimuovi la parentesi aperta
        if (ops.length === 0 || ops.pop() !== '(') {
          throw new SyntaxError("Parentesi non bilanciate.");
        }
      }
    } else if (typeof token === 'object' && 'op' in token) {
      // È un operatore (binario o unario)
      const op = token as Operator;

      // Gestione degli operatori unari (U+ e U-)
      if (op === 'U+' || op === 'U-') {
        // Gli operatori unari agiscono sul prossimo operando
        // Li trattiamo come se fossero operatori a precedenza molto alta
        // e li spingiamo direttamente nello stack degli operatori.
        ops.push(op);
      } else {
        // Operatori binari standard
        while (
          ops.length > 0 &&
          ops[ops.length - 1] !== '(' &&
          precedence(ops[ops.length - 1] as Operator) >= precedence(op)
        ) {
          const topOp = ops.pop() as Operator;
          if (values.length < 2) throw new SyntaxError("Operandi mancanti.");
          const val2 = values.pop()!;
          const val1 = values.pop()!;
          values.push(applyOp(topOp, val2, val1));
        }
        ops.push(op);
      }
    } else {
        // Questo caso dovrebbe essere coperto dalla tokenizzazione, ma è una safety check
        throw new SyntaxError("Token non valido.");
    }

    tokenIndex++;
  }

  // --- Fase 3: Pulizia e Risultato Finale ---

  // Esegui tutte le operazioni rimanenti nello stack
  while (ops.length > 0) {
    const op = ops.pop() as Operator;
    if (op === '(') {
      throw new SyntaxError("Parentesi non bilanciate.");
    }
    if (values.length < 2) throw new SyntaxError("Operandi mancanti.");
    const val2 = values.pop()!;
    const val1 = values.pop()!;
    values.push(applyOp(op, val2, val1));
  }

  // Verifica se è rimasto un solo risultato
  if (values.length !== 1) {
    throw new SyntaxError("Espressione incompleta o malformata.");
  }

  return values[0];
}
