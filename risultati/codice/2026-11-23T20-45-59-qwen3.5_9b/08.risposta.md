export class CacheLRU<K, V> {
  private capacity: number;
  private map: Map<K, Node<V>>;
  private head: Node<V>;
  private tail: Node<V>;
  private size: number;

  // Classe interna per i nodi della lista a doppio collegamento
  private class Node {
    key: K;
    value: V;
    prev: Node | null;
    next: Node | null;

    constructor(key: K, value: V) {
      this.key = key;
      this.value = value;
      this.prev = null;
      this.next = null;
    }
  }

  constructor(capacita: number) {
    if (capacita < 1) throw new RangeError("Capacità deve essere >= 1");
    this.capacity = capacita;
    this.map = new Map<K, Node<V>>();
    
    // Nodi fittizi per la lista a doppio collegamento
    this.head = { key: null as any, value: null as any, prev: null, next: null };
    this.tail = { key: null as any, value: null as any, prev: null, next: null };
    
    // Collegamento iniziale tra i nodi fittizi
    this.head.next = this.tail;
    this.tail.prev = this.head;
    
    this.size = 0;
  }

  get(chiave: K): V | undefined {
    const node = this.map.get(chiave);
    if (!node) return undefined;
    
    // Se la chiave esiste, sposta il nodo alla fine (più recente)
    this.moveToEnd(node);
    return node.value;
  }

    let node = this.map.get(chiave);
    
    if (node) {
      // Chiave esiste: aggiorna valore e sposta alla fine
      node.value = valore;
      this.moveToEnd(node);

      const prev = this.tail.prev;
      prev.next = node;
      node.prev = prev;
      node.next = this.tail;
      this.tail.prev = node;
      
      this.size++;

      lruNode.prev.next = lruNode.next;

    }
