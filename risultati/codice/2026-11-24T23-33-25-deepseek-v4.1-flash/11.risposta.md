function parseDate(s: string): number {
  if (!/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/.test(s)) {
    throw new RangeError("Formato data non valido");
  }

  const y = Number(s.slice(0, 4));
  const m = Number(s.slice(5, 7));
  const d = Number(s.slice(8, 10));

  if (y < 1990 || y > 2199) throw new RangeError("Anno fuori intervallo");
  if (m < 1 || m > 12) throw new RangeError("Mese non valido");
  if (d < 1 || d > 31) throw new RangeError("Giorno non valido");

  const t = Date.UTC(y, m - 1, d);
  const date = new Date(t);

  if (
    date.getUTCFullYear() !== y ||
    date.getUTCMonth() !== m - 1 ||
    date.getUTCDate() !== d
  ) {
    throw new RangeError("Data inesistente");
  }

  return t;
}

function easterSunday(year: number): { month: number; day: number } {
  const a = year % 19;
  const b = Math.floor(year / 100);
  const c = year % 100;
  const d = Math.floor(b / 4);
  const e = b % 4;
  const f = Math.floor((b + 8) / 25);
  const g = Math.floor((b - f + 1) / 3);
  const h = (19 * a + b - d - g + 15) % 30;
  const i = Math.floor(c / 4);
  const k = c % 4;
  const l = (32 + 2 * e + 2 * i - h - k) % 7;
  const m = Math.floor((a + 11 * h + 22 * l) / 451);
  const month = Math.floor((h + l - 7 * m + 114) / 31);
  const day = ((h + l - 7 * m + 114) % 31) + 1;

  return { month, day };
}
