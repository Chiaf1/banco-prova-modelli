export function aRomano(n: number): string {
  if (!Number.isInteger(n) || n < 1 || n > 3999) {
    throw new RangeError("n deve essere un intero tra 1 e 3999");
  }

  const valori: [number, string][] = [
    [1000, "M"],
    [900, "CM"],
    [500, "D"],
    [400, "CD"],
    [100, "C"],
    [90, "XC"],
    [50, "L"],
    [40, "XL"],
    [10, "X"],
    [9, "IX"],
    [5, "V"],
    [4, "IV"],
    [1, "I"],
  ];

  let risultato = "";
  for (const [valore, simbolo] of valori) {
    while (n >= valore) {
      risultato += simbolo;
      n -= valore;
    }
  }

  return risultato;
}

export function daRomano(s: string): number {
  if (typeof s !== "string") {
    throw new RangeError("s deve essere una stringa");
  }

  const valori: Record<string, number> = {
    I: 1,
    V: 5,
    X: 10,
    L: 50,
    C: 100,
    D: 500,
    M: 1000,
  };

  let totale = 0;

  for (let i = 0; i < s.length; i++) {
    const char = s.charAt(i);
    const valore = valori[char];

    if (valore === undefined) {
      throw new RangeError("Numero romano non valido");
    }

    const nextChar = s.charAt(i + 1);
    const prossimo = nextChar ? (valori[nextChar] ?? 0) : 0;

    if (valore < prossimo) {
      totale -= valore;
    } else {
      totale += valore;
    }
  }

  if (totale < 1 || totale > 3999) {
    throw new RangeError("Numero romano non valido");
  }

  if (aRomano(totale) !== s) {
    throw new RangeError("Numero romano non valido");
  }

  return totale;
}
