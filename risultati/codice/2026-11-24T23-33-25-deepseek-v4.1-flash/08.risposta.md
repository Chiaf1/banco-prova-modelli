export class CacheLRU<K, V> {
  #capacita: number;
  #mappa: Map<K, V>;

  constructor(capacita: number) {
    if (!Number.isInteger(capacita) || capacita < 1) {
      throw new RangeError("capacita deve essere un intero >= 1");
    }
    this.#capacita = capacita;
    this.#mappa = new Map<K, V>();
  }

  get(chiave: K): V | undefined {
    const mappa = this.#mappa;
    if (!mappa.has(chiave)) {
      return undefined;
    }

    const valore = mappa.get(chiave) as V;
    mappa.delete(chiave);
    mappa.set(chiave, valore);
    return valore;
  }

  set(chiave: K, valore: V): void {
    const mappa = this.#mappa;

    if (mappa.has(chiave)) {
      mappa.delete(chiave);
    } else if (mappa.size >= this.#capacita) {
      const menoRecente = mappa.keys().next().value as K;
      mappa.delete(menoRecente);
    }

    mappa.set(chiave, valore);
  }

  has(chiave: K): boolean {
    return this.#mappa.has(chiave);
  }

  get size(): number {
    return this.#mappa.size;
  }

  chiavi(): K[] {
    return Array.from(this.#mappa.keys());
  }
}
