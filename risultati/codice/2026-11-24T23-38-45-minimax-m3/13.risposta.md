export type Stato = "fermo" | "avvio" | "lavoro" | "pausa" | "allarme";

export type Esito = {
  stato: Stato;
  pezzi: number;
  gravita: number;
  confermato: boolean;
  rifiutati: { indice: number; evento: string; motivo: string }[];
};

export function simula(eventi: string[]): Esito {
  const noti = new Set([
    "start", "pronto", "pezzo", "pausa", "riprendi",
    "stop", "allarme", "conferma", "reset"
  ]);

  let stato: Stato = "fermo";
  let pezzi = 0;
  let gravita = 0;
  let confermato = false;
  let statoPrecedente: Stato = "fermo";
  const rifiutati: { indice: number; evento: string; motivo: string }[] = [];
  let rifiutiConsecutivi = 0;

  const vaiInAllarme = (da: Stato) => {
    statoPrecedente = da;
    stato = "allarme";
    gravita = 1;
    confermato = false;
    rifiutiConsecutivi = 0;
  };

  for (let i = 0; i < eventi.length; i++) {
    const ev = eventi[i];

    if (!noti.has(ev)) {
      rifiutati.push({ indice: i, evento: ev, motivo: "sconosciuto" });
      if (stato !== "allarme") {
        rifiutiConsecutivi++;
        if (rifiutiConsecutivi === 3) {
          vaiInAllarme(stato);
        }
      }
      continue;
    }

    if (stato === "allarme") {
      if (ev === "allarme") {
        gravita = gravita < 3 ? gravita + 1 : 3;
        confermato = false;
        rifiutiConsecutivi = 0;
      } else if (ev === "conferma") {
        if (!confermato) {
          confermato = true;
          rifiutiConsecutivi = 0;
        } else {
          rifiutati.push({ indice: i, evento: ev, motivo: "già confermato" });
        }
      } else if (ev === "riprendi") {
        if (!confermato) {
          rifiutati.push({ indice: i, evento: ev, motivo: "allarme da confermare" });
        } else if (gravita > 1) {
          rifiutati.push({ indice: i, evento: ev, motivo: "gravità troppo alta" });
        } else {
          stato = statoPrecedente;
          gravita = 0;
          confermato = false;
          rifiutiConsecutivi = 0;
        }
      } else if (ev === "reset") {
        if (!confermato) {
          rifiutati.push({ indice: i, evento: ev, motivo: "allarme da confermare" });
        } else {
          stato = "fermo";
          gravita = 0;
          confermato = false;
          rifiutiConsecutivi = 0;
        }
      } else {
        rifiutati.push({ indice: i, evento: ev, motivo: "non ammesso in allarme" });
      }
      continue;
    }

    // Fuori dall'allarme
    let accettato = false;
    if (ev === "start" && stato === "fermo") {
      stato = "avvio";
      accettato = true;
    } else if (ev === "pronto" && stato === "avvio") {
      stato = "lavoro";
      accettato = true;
    } else if (ev === "pezzo" && stato === "lavoro") {
      pezzi++;
      accettato = true;
    } else if (ev === "pausa" && stato === "lavoro") {
      stato = "pausa";
      accettato = true;
    } else if (ev === "riprendi" && stato === "pausa") {
      stato = "lavoro";
      accettato = true;
    } else if (ev === "stop" && (stato === "avvio" || stato === "lavoro" || stato === "pausa")) {
      stato = "fermo";
      accettato = true;
    } else if (ev === "allarme") {
      vaiInAllarme(stato);
      accettato = true;
    }

    if (accettato) {
      rifiutiConsecutivi = 0;
    } else {
      rifiutati.push({ indice: i, evento: ev, motivo: `non ammesso in ${stato}` });
      rifiutiConsecutivi++;
      if (rifiutiConsecutivi === 3) {
        vaiInAllarme(stato);
      }
    }
  }

  return { stato, pezzi, gravita, confermato, rifiutati };
}
