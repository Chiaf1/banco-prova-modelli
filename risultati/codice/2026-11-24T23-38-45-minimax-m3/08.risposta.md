interface Nodo<K, V> {
  chiave: K;
  valore: V;
  prev: Nodo<K, V> | null;
  next: Nodo<K, V> | null;
}

export class CacheLRU<K, V> {
  private readonly capacita: number;
  private readonly mappa: Map<K, Nodo<K, V>>;
  private head: Nodo<K, V> | null = null;
  private tail: Nodo<K, V> | null = null;

  constructor(capacita: number) {
    if (!Number.isInteger(capacita) || capacita < 1) {
      throw new RangeError("La capacità deve essere un intero ≥ 1");
    }
    this.capacita = capacita;
    this.mappa = new Map();
  }

  get(chiave: K): V | undefined {
    const nodo = this.mappa.get(chiave);
    if (nodo === undefined) {
      return undefined;
    }
    this.spostaInTesta(nodo);
    return nodo.valore;
  }

  set(chiave: K, valore: V): void {
    const nodoEsistente = this.mappa.get(chiave);
    if (nodoEsistente !== undefined) {
      nodoEsistente.valore = valore;
      this.spostaInTesta(nodoEsistente);
      return;
    }
    if (this.mappa.size >= this.capacita) {
      const vecchioTail = this.tail;
      if (vecchioTail !== null) {
        this.mappa.delete(vecchioTail.chiave);
        this.rimuoviNodo(vecchioTail);
      }
    }
    const nuovoNodo: Nodo<K, V> = { chiave, valore, prev: null, next: null };
    this.mappa.set(chiave, nuovoNodo);
    this.aggiungiInTesta(nuovoNodo);
  }

  has(chiave: K): boolean {
    return this.mappa.has(chiave);
  }

  get size(): number {
    return this.mappa.size;
  }

  chiavi(): K[] {
    const risultato: K[] = [];
    let corrente: Nodo<K, V> | null = this.tail;
    while (corrente !== null) {
      risultato.push(corrente.chiave);
      corrente = corrente.next;
    }
    return risultato;
  }

  private spostaInTesta(nodo: Nodo<K, V>): void {
    if (nodo === this.head) {
      return;
    }
    this.rimuoviNodo(nodo);
    this.aggiungiInTesta(nodo);
  }

  private rimuoviNodo(nodo: Nodo<K, V>): void {
    if (nodo.prev !== null) {
      nodo.prev.next = nodo.next;
    } else {
      this.tail = nodo.next;
    }
    if (nodo.next !== null) {
      nodo.next.prev = nodo.prev;
    } else {
      this.head = nodo.prev;
    }
    nodo.prev = null;
    nodo.next = null;
  }

  private aggiungiInTesta(nodo: Nodo<K, V>): void {
    nodo.prev = null;
    nodo.next = this.head;
    if (this.head !== null) {
      this.head.prev = nodo;
    } else {
      this.tail = nodo;
    }
    this.head = nodo;
  }
}
