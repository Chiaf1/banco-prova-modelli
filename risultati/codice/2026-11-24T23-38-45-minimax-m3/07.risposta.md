export function aRomano(n: number): string {
  if (!Number.isInteger(n) || n < 1 || n > 3999) {
    throw new RangeError("n deve essere un intero tra 1 e 3999");
  }
  const valori: [number, string][] = [
    [1000, "M"],
    [900, "CM"],
    [500, "D"],
    [400, "CD"],
    [100, "C"],
    [90, "XC"],
    [50, "L"],
    [40, "XL"],
    [10, "X"],
    [9, "IX"],
    [5, "V"],
    [4, "IV"],
    [1, "I"],
  ];
  let risultato = "";
  for (const [valore, simbolo] of valori) {
    while (n >= valore) {
      risultato += simbolo;
      n -= valore;
    }
  }
  return risultato;
}

export function daRomano(s: string): number {
  if (typeof s !== "string" || s.length === 0) {
    throw new RangeError("stringa non valida");
  }
  const valori: Record<string, number> = {
    I: 1,
    V: 5,
    X: 10,
    L: 50,
    C: 100,
    D: 500,
    M: 1000,
  };
  for (const c of s) {
    if (valori[c] === undefined) {
      throw new RangeError("stringa non valida");
    }
  }
  let n = 0;
  for (let i = 0; i < s.length; i++) {
    const curr = valori[s[i]];
    const next = i + 1 < s.length ? valori[s[i + 1]] : 0;
    if (curr < next) {
      n -= curr;
    } else {
      n += curr;
    }
  }
  if (n < 1 || n > 3999) {
    throw new RangeError("stringa non valida");
  }
  if (aRomano(n) !== s) {
    throw new RangeError("stringa non valida");
  }
  return n;
}
