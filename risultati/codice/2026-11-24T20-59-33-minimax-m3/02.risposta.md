export function palindromo(testo: string): boolean {
  // Tieni solo lettere minuscole e cifre, ignorando tutto il resto
  const pulito = testo.toLowerCase().replace(/[^a-z0-9]/g, "");

  // Confronto con due puntatori
  let sinistra = 0;
  let destra = pulito.length - 1;
  while (sinistra < destra) {
    if (pulito[sinistra] !== pulito[destra]) {
      return false;
    }
    sinistra++;
    destra--;
  }

  return true;
}
