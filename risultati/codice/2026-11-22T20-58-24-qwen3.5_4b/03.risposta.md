export function fizzBuzz(n: number): string[] {
  if (!Number.isInteger(n) || n < 0) {
    throw new RangeError("n deve essere un intero non negativo");
  }

  const result: string[] = [];
  for (let i = 1; i <= n; i++) {
    let str = "";
    if (i % 3 === 0 && i % 5 === 0) {
      str = "FizzBuzz";
    } else if (i % 3 === 0) {
      str = "Fizz";
    } else if (i % 5 === 0) {
      str = "Buzz";
    } else {
      str = i.toString();
    }
    result.push(str);
  }
  return result;
}
