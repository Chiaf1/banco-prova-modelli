export function fondiIntervalli(intervalli: [number, number][]): [number, number][] {
  if (intervalli.length === 0) return [];

  const copia: [number, number][] = [];

  for (const [inizio, fine] of intervalli) {
    if (!Number.isFinite(inizio) || !Number.isFinite(fine)) {
      throw new RangeError("Intervallo con valori non finiti");
    }
    if (inizio > fine) {
      throw new RangeError("Intervallo con inizio maggiore della fine");
    }
    copia.push([inizio, fine]);
  }

  copia.sort((a, b) => a[0] - b[0] || a[1] - b[1]);

  const risultato: [number, number][] = [];

  for (const [inizio, fine] of copia) {
    if (risultato.length === 0) {
      risultato.push([inizio, fine]);
      continue;
    }

    const ultimo = risultato[risultato.length - 1];

    if (inizio <= ultimo[1]) {
      if (fine > ultimo[1]) {
        ultimo[1] = fine;
      }
    } else {
      risultato.push([inizio, fine]);
    }
  }

  return risultato;
}
