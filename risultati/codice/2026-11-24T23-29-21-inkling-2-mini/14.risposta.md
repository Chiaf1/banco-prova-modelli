export function diffRighe(vecchio: string, nuovo: string, contesto: number = 1): string {
  if (!Number.isInteger(contesto) || contesto < 0) {
    throw new RangeError("contesto deve essere un intero >= 0");
  }

  const oldLines = parseLines(vecchio);
  const newLines = parseLines(nuovo);
  const n = oldLines.length;
  const m = newLines.length;

  // cost[i][j] = minimo numero di cancellazioni+inserimenti per trasformare old[i:] in new[j:]
  const cost: number[][] = Array.from({ length: n + 1 }, () => new Array<number>(m + 1).fill(0));
  for (let i = n; i >= 0; i--) cost[i][m] = n - i;
  for (let j = m; j >= 0; j--) cost[n][j] = m - j;
  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      if (oldLines[i] === newLines[j]) {
        cost[i][j] = cost[i + 1][j + 1];
      } else {
        cost[i][j] = 1 + Math.min(cost[i + 1][j], cost[i][j + 1]);
      }
    }
  }

  type Op = { t: "eq" | "del" | "ins"; i: number; j: number };
  const script: Op[] = [];
  let i = 0;
  let j = 0;
  while (i < n || j < m) {
    if (i < n && j < m && oldLines[i] === newLines[j]) {
      script.push({ t: "eq", i, j });
      i++;
      j++;
    } else if (i < n && 1 + cost[i + 1][j] === cost[i][j]) {
      script.push({ t: "del", i, j });
      i++;
    } else {
      script.push({ t: "ins", i, j });
      j++;
    }
  }

  const nonEq: number[] = [];
  for (let k = 0; k < script.length; k++) {
    if (script[k].t !== "eq") nonEq.push(k);
  }
  if (nonEq.length === 0) return "";

  const blocks: { start: number; end: number }[] = [];
  for (const idx of nonEq) {
    const last = blocks[blocks.length - 1];
    if (last && idx - last.end - 1 <= 2 * contesto) {
      last.end = idx;
    } else {
      blocks.push({ start: idx, end: idx });
    }
  }

  const out: string[] = [];
  for (const b of blocks) {
    let s = b.start;
    let e = b.end;
    let before = 0;
    while (s > 0 && script[s - 1].t === "eq" && before < contesto) {
      s--;
      before++;
    }
    let after = 0;
    while (e < script.length - 1 && script[e + 1].t === "eq" && after < contesto) {
      e++;
      after++;
    }

    let oldBefore = 0;
    let newBefore = 0;
    for (let k = 0; k < s; k++) {
      const t = script[k].t;
      if (t === "eq" || t === "del") oldBefore++;
      if (t === "eq" || t === "ins") newBefore++;
    }

    let N = 0;
    let M = 0;
    for (let k = s; k <= e; k++) {
      const t = script[k].t;
      if (t === "eq") {
        N++;
        M++;
      } else if (t === "del") {
        N++;
      } else {
        M++;
      }
    }

    const A = oldBefore + 1;
    const B = newBefore + 1;
    out.push(`@@ -${A},${N} +${B},${M} @@\n`);
    for (let k = s; k <= e; k++) {
      const op = script[k];
      if (op.t === "eq") out.push(" " + oldLines[op.i] + "\n");
      else if (op.t === "del") out.push("-" + oldLines[op.i] + "\n");
      else out.push("+" + newLines[op.j] + "\n");
    }
  }

  return out.join("");
}

function parseLines(t: string): string[] {
  if (t.length === 0) return [];
  if (t.endsWith("\n")) return t.slice(0, -1).split("\n");
  return t.split("\n");
}
