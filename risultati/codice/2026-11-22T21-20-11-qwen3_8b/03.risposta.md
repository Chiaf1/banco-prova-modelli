export function fizzBuzz(n: number): string[] {
  if (n < 1) {
    if (n === 0) {
      return [];
    } else {
      throw new RangeError("n must be positive");
    }
  }
  if (!Number.isInteger(n)) {
    throw new RangeError("n must be an integer");
  }
  const result: string[] = [];
  for (let i = 1; i <= n; i++) {
    if (i % 3 === 0 && i % 5 === 0) {
      result.push("FizzBuzz");
    } else if (i % 3 === 0) {
      result.push("Fizz");
    } else if (i % 5 === 0) {
      result.push("Buzz");
    } else {
      result.push(i.toString());
    }
  }
  return result;
}
