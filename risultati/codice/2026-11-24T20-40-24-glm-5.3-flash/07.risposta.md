export function aRomano(n: number): string {
  if (!Number.isInteger(n) || n < 1 || n > 3999) {
    throw new RangeError("numero non valido: " + n);
  }
  const migliaia = ["", "M", "MM", "MMM"];
  const centinaia = ["", "C", "CC", "CCC", "CD", "D", "DC", "DCC", "DCCC", "CM"];
  const decine = ["", "X", "XX", "XXX", "XL", "L", "LX", "LXX", "LXXX", "XC"];
  const unita = ["", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX"];

  const migliaiaPart = Math.floor(n / 1000);
  const centinaiaPart = Math.floor((n % 1000) / 100);
  const decinePart = Math.floor((n % 100) / 10);
  const unitaPart = n % 10;

  return migliaia[migliaiaPart] + centinaia[centinaiaPart] + decine[decinePart] + unita[unitaPart];
}

const valori: Record<string, number> = {
  I: 1, V: 5, X: 10, L: 50, C: 100, D: 500, M: 1000,
};

// Forme standard per ogni posizione (unità, decine, centinaia)
const unitaForme = ["", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX"];
const decineForme = ["", "X", "XX", "XXX", "XL", "L", "LX", "LXX", "LXXX", "XC"];
const centinaiaForme = ["", "C", "CC", "CCC", "CD", "D", "DC", "DCC", "DCCC", "CM"];
const migliaiaForme = ["", "M", "MM", "MMM"];

const regexStandard = new RegExp(
  "^(" + migliaiaForme.join("|") + ")(" + centinaiaForme.join("|") + ")(" +
  decineForme.join("|") + ")(" + unitaForme.join("|") + ")$"
);

export function daRomano(s: string): number {
  if (typeof s !== "string") {
    throw new RangeError("input non valido");
  }
  const match = regexStandard.exec(s);
  if (match === null) {
    throw new RangeError("numero romano non valido: " + s);
  }

  let totale = 0;
  for (const char of s) {
    const v = valori[char];
    totale += v;
  }
  // Correzione per le sottrazioni: ogni coppia di sottrazione
  // (IV, IX, XL, XC, CD, CM) è stata sommata con il valore sbagliato
  const coppie: [string, number][] = [
    ["IV", 2], ["IX", 2], ["XL", 20], ["XC", 20], ["CD", 200], ["CM", 200],
  ];
  for (const [coppia, correzione] of coppie) {
    if (s.includes(coppia)) {
      totale -= correzione;
    }
  }
  return totale;
}
