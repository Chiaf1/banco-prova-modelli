export function palindromo(testo: string): boolean {
  // Filtra solo lettere e cifre ASCII, convertendo in minuscolo
  const filtrato = testo
    .toLowerCase()
    .replace(/[^a-z0-9]/g, '');
  
  // Un testo senza lettere né cifre è palindromo
  if (filtrato.length === 0) {
    return true;
  }
  
  // Confronta la stringa filtrata con la sua inversa
  const invertito = filtrato.split('').reverse().join('');
  return filtrato === invertito;
}
