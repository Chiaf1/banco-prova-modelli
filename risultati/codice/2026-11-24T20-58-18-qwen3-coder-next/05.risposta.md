export function raggruppa<T>(elementi: T[], chiave: (elemento: T) => string): Record<string, T[]> {
  const risultato: Record<string, T[]> = {};
  const chiaviOrdine: string[] = [];

  for (const elemento of elementi) {
    const chiaveElemento = chiave(elemento);
    
    if (!(chiaveElemento in risultato)) {
      risultato[chiaveElemento] = [];
      chiaviOrdine.push(chiaveElemento);
    }
    
    risultato[chiaveElemento].push(elemento);
  }

  // Costruisce un oggetto con le chiavi nell'ordine in cui sono state incontrate
  const oggettoRisultato: Record<string, T[]> = {};
  for (const k of chiaviOrdine) {
    oggettoRisultato[k] = risultato[k];
  }

  return oggettoRisultato;
}
