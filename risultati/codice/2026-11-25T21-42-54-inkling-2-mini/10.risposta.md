export function valuta(espressione: string): number {
  type Token =
    | { tipo: "numero"; valore: number }
    | { tipo: "operatore"; simbolo: string }
    | { tipo: "aperta" }
    | { tipo: "chiusa" };

  const token: Token[] = [];
  let i = 0;
  const n = espressione.length;

  while (i < n) {
    const c = espressione[i];
    if (c === " " || c === "\t" || c === "\n" || c === "\r") {
      i++;
      continue;
    }
    if ((c >= "0" && c <= "9") || c === ".") {
      const m = /^(?:[0-9]+(?:\.[0-9]*)?|\.[0-9]+)/.exec(espressione.slice(i));
      if (m === null) {
        throw new SyntaxError("numero malformato");
      }
      token.push({ tipo: "numero", valore: Number(m[0]) });
      i += m[0].length;
      continue;
    }
    if (c === "+" || c === "-" || c === "*" || c === "/") {
      token.push({ tipo: "operatore", simbolo: c });
      i++;
      continue;
    }
    if (c === "(") {
      token.push({ tipo: "aperta" });
      i++;
      continue;
    }
    if (c === ")") {
      token.push({ tipo: "chiusa" });
      i++;
      continue;
    }
    throw new SyntaxError("carattere sconosciuto: " + c);
  }

  let pos = 0;

  function peek(): Token | undefined {
    return token[pos];
  }

  function error(messaggio: string): never {
    throw new SyntaxError(messaggio);
  }

  function parseExpression(): number {
    let valore = parseTerm();
    for (;;) {
      const t = peek();
      if (t !== undefined && t.tipo === "operatore" && (t.simbolo === "+" || t.simbolo === "-")) {
        pos++;
        const rhs = parseTerm();
        valore = t.simbolo === "+" ? valore + rhs : valore - rhs;
      } else {
        return valore;
      }
    }
  }

  function parseTerm(): number {
    let valore = parseUnary();
    for (;;) {
      const t = peek();
      if (t !== undefined && t.tipo === "operatore" && (t.simbolo === "*" || t.simbolo === "/")) {
        pos++;
        const rhs = parseUnary();
        valore = t.simbolo === "*" ? valore * rhs : valore / rhs;
      } else {
        return valore;
      }
    }
  }

  function parseUnary(): number {
    const t = peek();
    if (t !== undefined && t.tipo === "operatore" && (t.simbolo === "-" || t.simbolo === "+")) {
      pos++;
      const v = parseUnary();
      return t.simbolo === "-" ? -v : v;
    }
    return parsePrimary();
  }

  function parsePrimary(): number {
    const t = peek();
    if (t === undefined) {
      error("fine dell'espressione");
    }
    if (t.tipo === "numero") {
      pos++;
      return t.valore;
    }
    if (t.tipo === "aperta") {
      pos++;
      const v = parseExpression();
      const c = peek();
      if (c === undefined || c.tipo !== "chiusa") {
        error("parentesi chiusa mancante");
      }
      pos++;
      return v;
    }
    error("carattere inatteso");
  }

  const risultato = parseExpression();
  if (pos < token.length) {
    error("caratteri inattesi alla fine dell'espressione");
  }
  return risultato;
}
