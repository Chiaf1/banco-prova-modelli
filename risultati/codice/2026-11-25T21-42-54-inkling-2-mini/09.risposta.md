export function fondiIntervalli(intervalli: [number, number][]): [number, number][] {
  for (const intervallo of intervalli) {
    if (!Array.isArray(intervallo)) {
      throw new RangeError("Intervallo non valido");
    }
    const [inizio, fine] = intervallo;
    if (
      typeof inizio !== "number" ||
      typeof fine !== "number" ||
      !Number.isFinite(inizio) ||
      !Number.isFinite(fine)
    ) {
      throw new RangeError("Valori non finiti o non numerici");
    }
    if (inizio > fine) {
      throw new RangeError("Inizio maggiore di fine");
    }
  }

  const ordinati = [...intervalli].sort((a, b) => a[0] - b[0] || a[1] - b[1]);

  const risultato: [number, number][] = [];
  for (const [inizio, fine] of ordinati) {
    const ultimo = risultato[risultato.length - 1];
    if (ultimo && inizio <= ultimo[1]) {
      if (fine > ultimo[1]) {
        ultimo[1] = fine;
      }
    } else {
      risultato.push([inizio, fine]);
    }
  }
  return risultato;
}
